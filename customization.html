Personnalisation

Palettes de couleurs

Palettes seaborn intégrées

df = pd.DataFrame({
    'x': range(10),
    'A': np.random.randn(10).cumsum(),
    'B': np.random.randn(10).cumsum(),
    'C': np.random.randn(10).cumsum(),
    'D': np.random.randn(10).cumsum(),
})

(Chart(df)
    .set_colors('deep')
    .line('x', ['A', 'B', 'C', 'D'], marker='o')
    .title("Palette seaborn 'deep'")
    .set_legend()
    .grid(alpha=0.3)
    .tight_layout()
    .show())

Palettes seaborn qualitatives disponibles :

Palette Usage recommandé
deep Catégories distinctes (défaut seaborn)
pastel Fond clair, couleurs douces
bright Forte visibilité
dark Fond sombre ou impression
colorblind Accessible aux daltoniens
Set1, Set2, Set3 Catégories qualitatives
tab10, tab20 Nombreuses catégories

Palette personnalisée

df = pd.DataFrame({
    'categorie': ['A', 'B', 'C', 'D', 'E'],
    'valeur':    [80, 65, 90, 45, 72],
})

(Chart(df)
    .create_custom_palette('brand', ['#003049','#d62828','#f77f00','#fcbf49','#eae2b7'])
    .set_colors('brand')
    .bar('categorie', 'valeur')
    .title("Palette personnalisée 'brand'")
    .labels(xlabel='Catégorie', ylabel='Valeur')
    .grid(alpha=0.3)
    .tight_layout()
    .show())
# Créer et réutiliser une palette personnalisée
chart.create_custom_palette(
    'brand',
    ['#003049', '#d62828', '#f77f00', '#fcbf49', '#eae2b7']
)
# Appliquer dans le même chaînage
chart.set_colors('brand')

Dégradé entre deux couleurs

df_dummy = pd.DataFrame({'x': [1], 'y': [1]})
(Chart(df_dummy)
    .generate_gradient_palette('ocean_fire', '#003049', '#d62828', n_colors=8)
    .preview_palette('ocean_fire', n_colors=8)
    .tight_layout()
    .show())
# Générer un dégradé de 10 teintes entre deux couleurs
chart.generate_gradient_palette('ocean_fire', '#003049', '#d62828', n_colors=10)
chart.set_colors('ocean_fire')

Prévisualiser une palette

df_dummy = pd.DataFrame({'x': [1], 'y': [1]})
(Chart(df_dummy)
    .preview_palette('viridis', n_colors=10)
    .tight_layout()
    .show())

Polices

Changer la police globale

np.random.seed(0)
df = pd.DataFrame({'x': range(8), 'y': np.random.randn(8).cumsum()})

(Chart(df)
    .font('Georgia', size=12)
    .line('x', 'y', marker='s', linewidth=2)
    .title('Titre en Georgia', fontsize=16)
    .labels(xlabel='Axe X', ylabel='Axe Y')
    .grid(alpha=0.3)
    .tight_layout()
    .show())
# Définir la police globale
chart.font(family='Times New Roman', size=13, weight='bold')

Lister les polices installées

from plutovizu import get_fonts

# Toutes les polices
get_fonts()

# Filtrer par nom
get_fonts(pattern='Arial')

# Depuis une instance
chart.get_available_fonts(pattern='Times')

# Affichage console avec limite
Chart.print_available_fonts(limit=20)
Chart.print_available_fonts(pattern='Calibri')

Styles matplotlib

np.random.seed(42)
df = pd.DataFrame({'x': range(10), 'y': np.random.randn(10).cumsum()})

(Chart(df)
    .set_style('ggplot')          # à appeler avant de tracer
    .line('x', 'y', linewidth=2)
    .title("Style 'ggplot'")
    .labels(xlabel='X', ylabel='Y')
    .tight_layout()
    .show())

set_theme()   # un style matplotlib reste actif : on revient au thème par défaut (editorial)

Comme un thème, un style matplotlib s’applique à tous les graphiques suivants. Pour l’essentiel, préférez les thèmes plutovizu, qui comprennent aussi les styles seaborn.

Styles populaires : 'ggplot', 'seaborn-v0_8', 'dark_background', 'bmh', 'fivethirtyeight'.

# Lister tous les styles installés
import matplotlib.pyplot as plt
print(plt.style.available)

Annotations

df = pd.DataFrame({
    'mois':   list(range(1, 13)),
    'ventes': [12, 15, 18, 22, 28, 35, 40, 38, 30, 25, 20, 16],
})

(Chart(df)
    .line('mois', 'ventes', marker='o', linewidth=2)
    .add_annotation(7, 40, 'Pic estival', fontsize=11,
                    color='red', arrow=True)
    .title('Ventes mensuelles avec annotation')
    .labels(xlabel='Mois', ylabel='k€')
    .grid(alpha=0.3)
    .tight_layout()
    .show())
chart.add_annotation(
    x=7, y=40,
    text='Pic estival',
    fontsize=11,
    color='red',
    arrow=True,
    arrowprops=dict(arrowstyle='->', color='red', lw=1.5)
)

Grille, légende, limites

np.random.seed(0)
df = pd.DataFrame({
    'x': range(20),
    'A': np.random.randn(20).cumsum(),
    'B': np.random.randn(20).cumsum(),
})

(Chart(df)
    .line('x', ['A', 'B'])
    .grid(show=True, alpha=0.4, linestyle='--', linewidth=0.8)
    .set_legend(show=True, location='upper left', fontsize=11, frameon=True)
    .set_limits(xlim=(0, 19), ylim=(-8, 8))
    .title("Grille personnalisée + limites d'axes")
    .tight_layout()
    .show())

Export

# Enregistrer en PNG haute résolution (retourne self — chaînable)
chart.save('sortie.png', dpi=300)

# Enregistrer en PDF vectoriel
chart.save('rapport.pdf', format='pdf')

# Enregistrer en SVG
chart.save('figure.svg', format='svg', bbox_inches='tight')

# Accéder aux objets matplotlib bruts pour personnalisation avancée
fig, ax = chart.get_figure()
ax.axvspan(3, 5, alpha=0.1, color='red')   # zone colorée
fig.savefig('custom.png', dpi=300)
